class Solution {
public:
    int countNonMinimum(vector<int>& nums) {
        // write your code here 
        
        
        
        if(nums.size()==1){            //edge case
            return 0;
        }
        
        int mini = INT_MAX;
        
        for(int i=0;i<nums.size();i++){    // miniminum val in the arr
            mini = min(mini,nums[i]);
        }
        
        int count =0;
        for(int i=0;i<nums.size();i++){    // count value greater than mini
            if(nums[i]!= mini){
                count++;
            }
        }
        
        return count;
    
    
    }
};
